PROGRAMMATIC USAGE
──────────────────────────────────────────────────────────────────────────────

from breeding_vat.modules.multi_advisor import MultiAdvisor
from breeding_vat.modules.gguf_loader import GGUFModelLoader

# Initialize
ma = MultiAdvisor.create_default_setup()  # Quick + Thorough
gguf = GGUFModelLoader()

# List GGUF models
for model in gguf.get_available_models():
    print(f"  {model['display_name']} - {model['quantization']}")

# Consult advisor
response = ma.consult_advisor(
    slot_name="quick",
    goal="Reasoning at 3B",
    base_models=["Qwen/Qwen2.5-1.5B", "deepseek-ai/DeepSeek-R1-Distill-Qwen-1.5B"],
    available_methods=["slerp", "ties", "task_arithmetic"]
)
print(response)

# Cleanup
ma.cleanup()
